{{- /* Internal cross-reference. Resolves with site.GetPage, which handles regular pages, top-level sections and nested sections alike. The previous implementation ranged over `union .Site.RegularPages .Site.Sections` and matched on a URL suffix; that missed sections unpredictably (/themes and /expl/bible_ref among them) and, worse, emitted nothing at all when it missed - leaving a hole in the sentence with no error anywhere. A miss now warns during the build, naming the file, and still renders the link text so the sentence stays readable. */ -}} {{- $val := .Get "val" -}} {{- $link := .Get "link" -}} {{- $parts := split $link "#" -}} {{- $path := index $parts 0 -}} {{- $fragment := "" -}} {{- if gt (len $parts) 1 -}} {{- $fragment = index (split (index $parts 1) "/") 0 -}} {{- end -}} {{- with site.GetPage $path -}} {{- $href := .RelPermalink -}} {{- with $fragment -}} {{- $href = printf "%s#%s" $href . -}} {{- end -}} {{- $val -}} {{- else -}} {{- $where := $.Page.RelPermalink -}} {{- with $.Page.File -}}{{- $where = .Path -}}{{- end -}} {{- warnf "int_link: no page found for %q (used in %s)" $link $where -}} {{- $val -}} {{- end -}}
