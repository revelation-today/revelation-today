{{- /* Label for one prev/next arrow. Kids lesson files are named NN-slug, and their titles carry a Bible reference rather than a number ("Rev. 17-18: Babylon Falls"), which says nothing about where in the series you are heading. So every numbered kids lesson shows "Lesson 15" on its own. Any other page keeps its plain title; the full title is still on the link's title attribute. */ -}} {{- $page := .page -}} {{- $lesson := .lesson -}} {{- $label := $page.Title -}} {{- if and $page.Weight (in $page.RelPermalink "/kids/") -}} {{- with $page.File -}} {{- if findRE "^[0-9]{2}-" .BaseFileName -}} {{- $label = printf "%s %d" $lesson $page.Weight -}} {{- end -}} {{- end -}} {{- end -}} {{- $label -}}
